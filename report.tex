\section{Denavit--Hartenberg Convention}

\subsection{Robot Description}
The KUKA LBR Med 7 R800 is a 7-DOF redundant surgical manipulator with all revolute joints. Its kinematic structure consists of alternating perpendicular joint axes, giving a 7-DOF configuration for a 6-DOF task space, with one degree of intrinsic redundancy.

\subsection{Frame Assignment (Standard DH Convention)}
Following the standard DH frame assignment procedure used in class:
\begin{enumerate}
    \item Number joint axes $z_0, \dots, z_6$ where $z_i$ is the axis of joint $i + 1$.
    \item Place Frame 0 at the base with $z_0$ pointing vertically upward.
    \item For $i = 1, \dots, 6$, place origin $O_i$ at the intersection of $z_{i-1}$ and $z_i$.
    \item Define $x_i$ along the common normal from $z_{i-1}$ to $z_i$, directed from joint $i$ to joint $i + 1$.
    \item Define $y_i$ to complete the right-handed frame.
    \item Define Frame 7 at the end-effector with $z_7$ aligned with $z_6$ and $O_7$ at the flange center.
\end{enumerate}

\textbf{Key structural observation:} All consecutive joint axes intersect, so all link offsets satisfy $a_i = 0$.

\section{Inverse Kinematics (Closed-Form)}

\subsection{Kinematic Decoupling Approach}
The robot has a spherical wrist formed by joints 5, 6, and 7: axes $z_4$, $z_5$, and $z_6$ intersect at point $O_6$. This enables kinematic decoupling:
\begin{itemize}
    \item Arm problem (joints 1--4): position the wrist center $O_6$.
    \item Wrist problem (joints 5--7): orient the end-effector.
\end{itemize}
\textbf{Redundancy resolution:} The robot has 7 DOF for a 6-DOF task. This report fixes $q_3 = 0$ to obtain a unique solution, reducing the arm problem to 3 unknowns $(q_1, q_2, q_4)$.

\subsection{Closed-Form Solution}

\subsubsection*{Step 1 -- Wrist center}
The wrist center is the origin of frame 6, offset from the end-effector along the approach vector $a_e = R_d[:,3]$:
\begin{equation}
    p_W = p_e - d_7\, a_e
\end{equation}

\subsubsection*{Step 2 -- Base rotation $q_1$}
\begin{equation}
    q_1 = \operatorname{atan2}(p_{Wy},\; p_{Wx})
    \quad \text{(or } q_1 + \pi \text{ for the back-arm branch)}
\end{equation}

\subsubsection*{Step 3 -- Elbow angle $q_4$}
Projecting the wrist center into the sagittal plane gives:
\begin{equation}
    X = p_{Wx}\cos q_1 + p_{Wy}\sin q_1, \qquad Z = p_{Wz} - d_1
\end{equation}
From the law of cosines applied to the triangle $O_2$--$O_4$--$O_6$:
\begin{equation}
    \cos q_4 = \frac{X^2 + Z^2 - d_3^2 - d_5^2}{2 d_3 d_5}
\end{equation}
Two solutions arise from $\sin q_4 = \pm\sqrt{1 - \cos^2 q_4}$ (elbow-up and elbow-down).

\subsubsection*{Step 4 -- Shoulder angle $q_2$}
Expanding the sagittal-plane geometry yields the linear system
\begin{equation}
    \begin{bmatrix} -(d_3 + d_5 c_4) & d_5 s_4 \\ d_5 s_4 & d_3 + d_5 c_4 \end{bmatrix}
    \begin{bmatrix} s_2 \\ c_2 \end{bmatrix}
    =
    \begin{bmatrix} X \\ Z \end{bmatrix}
\end{equation}
which gives $q_2 = \operatorname{atan2}(s_2, c_2)$ directly.

\subsubsection*{Steps 5--7 -- Wrist angles $q_5$, $q_6$, $q_7$}
With $q_1$--$q_4$ known, the remaining rotation is isolated as:
\begin{equation}
    R_{47} = \bigl(R_0^4\bigr)^\top R_d
\end{equation}
where $R_0^4$ is the rotation part of $A_1 A_2 A_3 A_4$. Expanding the symbolic product $A_5 A_6 A_7$ (with $\alpha_5 = +\pi/2$, $\alpha_6 = -\pi/2$, $\alpha_7 = 0$) yields:
\begin{equation}
    R_{47} =
    \begin{bmatrix}
        c_5 c_6 c_7 - s_5 s_7 & -c_5 c_6 s_7 - s_5 c_7 & -c_5 s_6 \\
        s_5 c_6 c_7 + c_5 s_7 & -s_5 c_6 s_7 + c_5 c_7 & -s_5 s_6 \\
        s_6 c_7               & -s_6 s_7               &  c_6
    \end{bmatrix}
\end{equation}
Reading off the entries:
\begin{align}
    q_6 &= \operatorname{atan2}\!\left(\sqrt{1 - R_{47}(3,3)^2},\; R_{47}(3,3)\right) \label{eq:q6}\\
    q_5 &= \operatorname{atan2}\!\left(-R_{47}(2,3),\; -R_{47}(1,3)\right) \quad \text{(when } \sin q_6 \neq 0\text{)} \label{eq:q5}\\
    q_7 &= \operatorname{atan2}\!\left(-R_{47}(3,2),\; R_{47}(3,1)\right) \label{eq:q7}
\end{align}
A second wrist branch is obtained by negating $\sin q_6$ in \eqref{eq:q6} and flipping the signs in \eqref{eq:q5}--\eqref{eq:q7}.
Combined with the two arm branches ($q_1$ front/back) and two elbow branches ($q_4$ up/down), the solver generates up to \textbf{8 closed-form solutions}, returning the one closest to a reference configuration.

\section{Geometric Jacobian}

\subsection{Definition}
For each revolute joint $i$, the geometric Jacobian columns are:
\begin{equation}
    J_{P_i} = z_{i-1} \times (p_e - O_{i-1}), \quad J_{O_i} = z_{i-1}
\end{equation}
This relates joint velocities to end-effector velocity: $v_e = J(q)\dot{q}$.

\subsection{Simulink Model}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.6\textwidth]{jacobian.png}
    \caption{Jacobian block in Simulink.}
    \label{fig:jacobian_block}
\end{figure}

\subsection{Validation -- Numerical Differentiation}
The Jacobian is validated via central finite differences:
\begin{equation}
    J_{P_i}^{\mathrm{num}} \approx \frac{p_e(q + \varepsilon e_i) - p_e(q - \varepsilon e_i)}{2\varepsilon}, \qquad \varepsilon = 10^{-7}
\end{equation}
The angular columns are extracted from the skew-symmetric matrix $S(\omega) = \dot{R}R^\top$ evaluated at the central difference.

\textbf{Interpretation.} When the rank drops from 6 to 5, one column of $J$ becomes a linear combination of the others: the robot loses the ability to produce end-effector velocity in the corresponding direction regardless of joint velocity. At such configurations the condition number diverges and the DLS pseudo-inverse (used in CLIK, Section~5) provides a regularised solution at the cost of small tracking error.

The wrist singularity (S3, $\sin q_6 = 0$) is the most commonly encountered: it occurs whenever joints 5 and 7 axes align ($z_4 \parallel z_6$), collapsing two rotational DOF into one. The elbow singularity (S2) is reached when the arm is fully extended or fully folded, eliminating one translational DOF.

\section{Closed Loop Inverse Kinematics (CLIK)}

\subsection{CLIK Law with Null-Space Stabilisation}
The CLIK control law using Damped Least Squares (DLS) is:
\begin{equation}
    \dot{q} = J^{\dagger}(q) \left( \begin{bmatrix} K_P e_P \\ K_O e_O \end{bmatrix} \right) + (I - J^{\dagger} J) \dot{q}_0
\end{equation}
where $J^{\dagger} = J^T (J J^T + k^2 I)^{-1}$ is the damped pseudo-inverse with $k = 10^{-3}$, $K_P = 10 I_3$, $K_O = 10 I_3$. The null-space term $\dot{q}_0 = K_n(q_{\mathrm{rest}} - q)$ with $K_n = 5$ stabilizes the redundancy towards a rest configuration. The desired trajectory is constant ($\dot{p}_d = 0$, $\omega_d = 0$), so only the error-feedback terms are active.

The orientation error is computed using the geometric cross-product formulation:
\begin{equation}
    e_O = \tfrac{1}{2}\bigl(n_e \times n_d + s_e \times s_d + a_e \times a_d\bigr)
\end{equation}
